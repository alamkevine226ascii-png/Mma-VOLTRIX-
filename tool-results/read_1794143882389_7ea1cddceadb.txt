    64→/**
    65→ * Mesure explicite du conteneur (ResizeObserver + ref callback) pour
    66→ * dimensionner les graphiques recharts sans ResponsiveContainer, dont la
    67→ * mesure initiale peut être faussée dans une grille fractionnaire.
    68→ */
    69→function useContainerSize<T extends HTMLElement>() {
    70→  const [size, setSize] = useState<{ w: number; h: number } | null>(null)
    71→  const [element, setElement] = useState<T | null>(null)
    72→
    73→  const setEl = useCallback((node: T | null) => {
    74→    setElement(node)
    75→  }, [])
    76→
    77→  useEffect(() => {
    78→    if (!element) return
    79→    const ro = new ResizeObserver((entries) => {
    80→      const r = entries[0].contentRect
    81→      setSize({ w: Math.round(r.width), h: Math.round(r.height) })
    82→    })
    83→    ro.observe(element)
    84→    return () => ro.disconnect()
    85→  }, [element])
    86→
    87→  return { setEl, size }
    88→}
